cmake_minimum_required(VERSION 3.22)
project(hello LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
find_package(OpenMP REQUIRED)
find_package(GTest REQUIRED)
add_library(calc calc.cpp)
add_executable(hello main.cpp)
target_link_libraries(hello PRIVATE calc OpenMP::OpenMP_CXX)
enable_testing()
add_executable(calc_test tests/calc_test.cpp)
target_include_directories(calc_test PRIVATE ${CMAKE_CURRENT_SOURCE_DIR})
target_link_libraries(calc_test PRIVATE calc GTest::gtest_main)
add_test(NAME calc_test COMMAND calc_test)
